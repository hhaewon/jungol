a+2b+3c = k (a,b,c >= 0) -> 해 개수 dp1[k]
2b+3c = k -> 해 개수 dp2[k]
3c = k -> 해 개수 dp3[k] : k % 3 == 0 : 1, k % 3 != 0 : 0

dp1[k]를 보자 (k >= 1)
a >= 1 -> dp1[k-1]
a = 0 -> 2b+3c = k -> dp2[k]
dp1[k] = dp1[k-1] + dp2[k]
dp1[0] = 1

dp2[k]를 보자
b >= 1 -> dp2[k-2]
b = 0 : dp3[k]
dp2[0] = 1
dp2[1] = 0

dp2[k] = dp3[k-2]+dp3[k]

dp1[4] = dp1[3] + dp2[4] = 3
dp1[3] = dp1[2] + dp2[3] = 2 + 1 = 3
dp1[2] = dp1[1] + dp2[2] = 1 + 1 = 2
dp1[1] = dp1[0] + dp2[1] = 1 + 0 = 1
dp2[2] = dp2[0] + dp3[2] = 1
dp2[3] = dp2[1] + dp3[3] = 1
dp2[4] = dp2[2]+dp3[4] = 0
